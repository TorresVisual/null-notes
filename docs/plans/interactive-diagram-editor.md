# Interactive Diagram Editor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (default) or superpowers:subagent-driven-development if explicitly requested.

**Goal:** Add a new `DiagramBlock` TipTap node backed by Excalidraw, giving notes an interactive box/arrow/connector diagram editor alongside the existing Mermaid (text) and ink (freehand) options.

**Architecture:** A new atom TipTap node stores two sidecar files per diagram — a `.excalidraw` JSON scene (editable source) and a `.svg` (cached static preview) — in a `NoteName.diagrams/` directory, mirroring `ink.rs`'s existing `NoteName.ink/` convention exactly. The note shows the static SVG inline; clicking it opens a full-viewport overlay mounting the real `<Excalidraw>` editor, and closing it re-exports the SVG and writes both files. Rust-side storage commands are a direct structural copy of `commands/ink.rs`.

**Tech Stack:** `@excalidraw/excalidraw` (v0.18, MIT, React 19-compatible — verified via `npm view`), Tauri IPC commands in Rust, TipTap 3 atom node + React node view (existing pattern from `InkBlock.ts`/`MermaidBlock.tsx`).

**Spec:** `docs/specs/interactive-diagram-editor-design.md`

## Global Constraints

- No `.drawio`/mxGraph XML support, no postMessage bridge to a real draw.io instance — Excalidraw only.
- No always-on inline canvas — click-to-open only, matching ink's click-to-draw model.
- Fully offline: Excalidraw fonts must be vendored locally, `window.EXCALIDRAW_ASSET_PATH` set — never fetched from a CDN.
- Mermaid (`MermaidBlock.tsx`) and ink (`InkBlock.ts`, `InkCanvas.tsx`) are untouched by this work.
- Follow existing sidecar-directory naming (`NoteName.<kind>/`) and Rust command style (`commands/ink.rs`) exactly — no `write_atomic`, no `resolve_relative` boundary check, since sidecar paths are always self-generated by the app, never attacker-supplied note text (same trust model `ink.rs` already relies on).

use std::path::Path;

fn diagrams_dir_for(note_path: &str) -> String {
    let p = Path::new(note_path);
    let stem = p.file_stem().unwrap_or_default().to_string_lossy();
    let dir = p.parent().unwrap_or(Path::new("."));
    dir.join(format!("{}.diagrams", stem))
        .to_string_lossy()
        .into_owned()
}

#[tauri::command]
pub async fn ensure_diagrams_dir(note_path: String) -> Result<String, String> {
    let dir = diagrams_dir_for(&note_path);
    tokio::fs::create_dir_all(&dir)
        .await
        .map_err(|e| e.to_string())?;
    Ok(dir)
}

#[tauri::command]
pub async fn read_diagram(path: String) -> Result<String, String> {
    tokio::fs::read_to_string(&path)
        .await
        .map_err(|e| e.to_string())
}

#[tauri::command]
pub async fn write_diagram(path: String, content: String) -> Result<(), String> {
    tokio::fs::write(&path, content)
        .await
        .map_err(|e| e.to_string())
}

Create `src/editor/extensions/DiagramEditorOverlay.tsx`:

```tsx
import { useEffect, useState } from 'react'
import { createPortal } from 'react-dom'
import { Excalidraw, exportToSvg } from '@excalidraw/excalidraw'
import '@excalidraw/excalidraw/index.css'
import type { ExcalidrawImperativeAPI, AppState, BinaryFiles } from '@excalidraw/excalidraw/types'
import type { ExcalidrawElement } from '@excalidraw/excalidraw/element/types'
import { diagramApi } from '../../lib/diagram'
import { report } from '../splitIntoTasks'

interface DiagramEditorOverlayProps {
  jsonSrc: string
  svgSrc: string
  onClose: (svg: string) => void
}

interface StoredScene {
  elements?: ExcalidrawElement[]
  appState?: Partial<AppState>
  files?: BinaryFiles
}

interface Scene {
  elements: ExcalidrawElement[]
  appState: Partial<AppState>
  files: BinaryFiles
}

export function DiagramEditorOverlay({ jsonSrc, svgSrc, onClose }: DiagramEditorOverlayProps) {
  const [scene, setScene] = useState<Scene | null>(null)
  const [api, setApi] = useState<ExcalidrawImperativeAPI | null>(null)

  useEffect(() => {
    let cancelled = false
    diagramApi
      .read(jsonSrc)
      .then((json) => {
        if (cancelled) return
        const parsed = JSON.parse(json) as StoredScene
        setScene({ elements: parsed.elements ?? [], appState: parsed.appState ?? {}, files: parsed.files ?? {} })
      })
      .catch(() => {
        if (!cancelled) setScene({ elements: [], appState: {}, files: {} })
      })
    return () => {
      cancelled = true
    }
  }, [jsonSrc])

  async function handleClose() {
    if (!api) {
      onClose('')
      return
    }
    const elements = api.getSceneElements()
    const appState = api.getAppState()
    const files = api.getFiles()
    const json = JSON.stringify({
      type: 'excalidraw',
      version: 2,
      source: 'null-notes',
      elements,
      appState,
      files,
    })
    const svgEl = await exportToSvg({ elements, appState, files })
    const svg = svgEl.outerHTML
    try {
      await diagramApi.write(jsonSrc, json)
      await diagramApi.write(svgSrc, svg)
    } catch (error) {
      report(error)
    }
    onClose(svg)
  }

  return createPortal(
    <div className="diagram-editor-overlay">
      <div className="diagram-editor-modal">
        <div className="diagram-editor-header">
          <button type="button" className="dialog-action-primary" onClick={() => void handleClose()}>
            Done
          </button>
        </div>
        <div className="diagram-editor-canvas">
          {scene && (
            <Excalidraw
              initialData={{ elements: scene.elements, appState: scene.appState, files: scene.files }}
              excalidrawAPI={(instance) => setApi(instance)}
            />
          )}
        </div>
      </div>
    </div>,
    document.body,
  )
}
```

```tsx
import { useEffect, useState } from 'react'
import { Node, mergeAttributes } from '@tiptap/core'
import { ReactNodeViewRenderer, NodeViewWrapper } from '@tiptap/react'
import type { ReactNodeViewProps } from '@tiptap/react'
import { diagramApi } from '../../lib/diagram'
import { DiagramEditorOverlay } from './DiagramEditorOverlay'

interface DiagramBlockAttrs {
  jsonSrc: string
  svgSrc: string
  width: number
  height: number
}

function DiagramBlockView({ node }: ReactNodeViewProps) {
  const { jsonSrc, svgSrc, width, height } = node.attrs as unknown as DiagramBlockAttrs
  const [svg, setSvg] = useState('')
  const [editing, setEditing] = useState(false)

  useEffect(() => {
    let cancelled = false
    diagramApi
      .read(svgSrc)
      .then((text) => {
        if (!cancelled) setSvg(text)
      })
      .catch(() => {
        if (!cancelled) setSvg('')
      })
    return () => {
      cancelled = true
    }
  }, [svgSrc])

  // insertDiagram (Task 5) dispatches this immediately after inserting a
  // brand-new, empty diagram — an empty scene has no useful static preview
  // to land on, so it jumps straight to the editor instead. Matched on
  // jsonSrc so only the node that was actually just inserted opens.
  useEffect(() => {
    function onAutoOpen(event: Event) {
      const detail = (event as CustomEvent<{ jsonSrc: string }>).detail
      if (detail.jsonSrc === jsonSrc) setEditing(true)
    }
    window.addEventListener('diagram:auto-open', onAutoOpen)
    return () => window.removeEventListener('diagram:auto-open', onAutoOpen)
  }, [jsonSrc])

  if (!jsonSrc) return <NodeViewWrapper className="diagram-block" />

  return (
    <NodeViewWrapper className="diagram-block" style={{ width, height }}>
      <div className="diagram-block-preview" dangerouslySetInnerHTML={{ __html: svg || '<svg></svg>' }} />
      <button type="button" className="diagram-block-edit" aria-label="Edit diagram" onClick={() => setEditing(true)}>
        Edit
      </button>
      {editing && (
        <DiagramEditorOverlay
          jsonSrc={jsonSrc}
          svgSrc={svgSrc}
          onClose={(newSvg) => {
            setSvg(newSvg)
            setEditing(false)
          }}
        />
      )}
    </NodeViewWrapper>
  )
}

export const DiagramBlock = Node.create({
  name: 'diagramBlock',
  group: 'block',
  atom: true,

  addAttributes() {
    return {
      jsonSrc: { default: null },
      svgSrc: { default: null },
      width: { default: 700 },
      height: { default: 500 },
    }
  },

  parseHTML() {
    return [
      {
        tag: 'div[data-type="diagram-block"]',
        getAttrs: (el) => ({
          jsonSrc: (el as HTMLElement).getAttribute('data-json-src'),
          svgSrc: (el as HTMLElement).getAttribute('data-svg-src'),
          width: parseInt((el as HTMLElement).getAttribute('data-width') ?? '700', 10),
          height: parseInt((el as HTMLElement).getAttribute('data-height') ?? '500', 10),
        }),
      },
    ]
  },

  renderHTML({ HTMLAttributes }) {
    return ['div', mergeAttributes(HTMLAttributes, { 'data-type': 'diagram-block' })]
  },

  addNodeView() {
    return ReactNodeViewRenderer(DiagramBlockView)
  },

  addStorage() {
    return {
      markdown: {
        serialize(state: Record<string, unknown>, node: { attrs: DiagramBlockAttrs }) {
          const { jsonSrc, svgSrc, width, height } = node.attrs
          const s = state as {
            write: (str: string) => void
            ensureNewLine: () => void
            closeBlock: (n: unknown) => void
          }
          s.write('```diagram\n')
          s.write(JSON.stringify({ jsonSrc, svgSrc, width, height }))
          s.ensureNewLine()
          s.write('```')
          s.closeBlock(node)
        },
        parse: {
          updateDOM(element: HTMLElement) {
            element.querySelectorAll<HTMLElement>('pre > code.language-diagram').forEach((code) => {
              const pre = code.parentElement
              if (!pre) return
              try {
                const data = JSON.parse(code.textContent ?? '{}') as Partial<DiagramBlockAttrs>
                const div = document.createElement('div')
                div.setAttribute('data-type', 'diagram-block')
                div.setAttribute('data-json-src', data.jsonSrc ?? '')
                div.setAttribute('data-svg-src', data.svgSrc ?? '')
                div.setAttribute('data-width', String(data.width ?? 700))
                div.setAttribute('data-height', String(data.height ?? 500))
                pre.replaceWith(div)
              } catch {
                // invalid JSON — leave as-is
              }
            })
          },
        },
      },
    }
  },
})
```

Create `src/editor/toolbar/insertDiagram.ts`:

```ts
import type { Editor } from '@tiptap/core'
import { diagramApi } from '../../lib/diagram'

export const DIAGRAM_WIDTH = 700
export const DIAGRAM_HEIGHT = 500

const EMPTY_SCENE_JSON = JSON.stringify({
  type: 'excalidraw',
  version: 2,
  source: 'null-notes',
  elements: [],
  appState: { viewBackgroundColor: '#ffffff' },
  files: {},
})

const EMPTY_SVG = '<svg xmlns="http://www.w3.org/2000/svg"></svg>'

// Shared by the toolbar's "Insert diagram" button and the slash command —
// both need the exact same sidecar-then-node sequence, matching insertInk's
// own reasoning for having one copy of this instead of two that could drift.
export async function insertDiagram(editor: Editor, notePath: string): Promise<void> {
  const dir = await diagramApi.ensureDir(notePath)
  const sep = dir.includes('\\') ? '\\' : '/'
  const id = Date.now()
  const jsonSrc = `${dir}${sep}diagram-${id}.excalidraw`
  const svgSrc = `${dir}${sep}diagram-${id}.svg`
  await diagramApi.write(jsonSrc, EMPTY_SCENE_JSON)
  await diagramApi.write(svgSrc, EMPTY_SVG)
  editor.commands.insertContent({
    type: 'diagramBlock',
    attrs: { jsonSrc, svgSrc, width: DIAGRAM_WIDTH, height: DIAGRAM_HEIGHT },
  })
  // A brand-new diagram has nothing worth previewing yet — open straight
  // into the editor instead of landing on an empty static block.
  window.dispatchEvent(new CustomEvent('diagram:auto-open', { detail: { jsonSrc } }))
}
```
